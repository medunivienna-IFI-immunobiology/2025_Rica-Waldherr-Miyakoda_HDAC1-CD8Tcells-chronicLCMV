# title: Cut&Run analysis - visualization of Hdac1, Runx3, H3K27ac, H3K27me3 enrichment at DARs in early-TexProg
# author: Monika Waldherr
# input: bed file of DARs from ATACseq, BigWig files of HDAC1, RUNX3, H3K27ac and H3K27me3 (WT and HDAC1-cKO)

# deeptools commands used

# get matrix
computeMatrix scale-regions -b 3000 -a 3000 --regionBodyLength 5000 -R TexProg_ATAC[genotype]open.bed -S [WT-HDAC1]rpkm.BigWig [KO-HDAC1]rpkm.BigWig [WT-RUNX3]rpkm.BigWig [KO-RUNX3]rpkm.BigWig [WT-H3K27ac]rpkm.BigWig [KO-H3K27ac]rpkm.BigWig [WT-H3K27me3]rpkm.BigWig [KO-H3K27me3]rpkm.BigWig --skipZeros -o [sample]_matrix_gene.mat.gz -p 30 --samplesLabel TprogWT-HDAC1 TprogKO-HDAC1 TprogWT-RUNX3 TprogKO-RUNX3 TprogWT-H3K27ac TprogKO-H3K27ac TprogWT-H3K27me3 TprogKO-H3K27me3 --smartLabels &

# plot heatmap
plotHeatmap -m [sample]_matrix_gene.mat.gz -out [sample]_heatmap.pdf --sortUsing sum --labelRotation 45 --colorMap inferno inferno Greens Greens Oranges Oranges Blues Blues --zMax 40 40 90 90 50 50 20 20 --yMax 40 40 90 90 50 50 20 20 &
